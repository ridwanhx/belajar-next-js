<!-- Eps. 04 - Styling -->

<!-- Secara default pada ekosistem Next.js akan menambahkan elemen styling dari globals.css -->
<!-- globals.css sendiri hanya bisa diimport dari _app.tsx -->
<!-- untuk membatasi lingkup styling untuk elemen-elemen tertentu, kita bisa memanfaatkan konsep CSS Module -->
<!-- CSS Module sendiri sifatnya adalah local scope (ketika digunakan, akan otomatis membuat unique classname) -->

<!-- untuk pemanggilan styles nya yang didefinisikan didalam component, kita bisa menuliskannya sebagai berikut: 
import alias from "@/styles/NamaModule.module.css

kemudian didalam div/pembungkus komponennya, kita bisa menambahkan className={alias.nama-class}
-->

<!-- untuk penempatan CSS Module nya itu sendiri, meskipun secara default Next.js akan menyarankan kita untuk menyimpan file module tersebut didalam folder styles, namun alangkah lebih rapi lagi untuk kita menyimpan module nya di satu folder yang sama di dalam folder komponen yang dimaksud -->

<!-- aturan tidak tertulis dan memang sebaiknya kita harus bisa beradaptasi dengan hal ini kedepannya ialah -->
<!-- bahwa penamaan kelas css pada ekosistem Next sebaiknya menggunakan camelCase -->
<!-- jangan lagi tulis nama-kelas, tapi sebaiknya namaKelas -->
<!-- kalaupun memang ingin mempertahankan gaya penulisan snake-case, nantinya pada saat pemanggilan style nya kita bisa menggunakan Bracket Notation, contoh:

import alias from './NamaModule.module.css'
... 
<div className={alias['nama-kelas']}>

sebenarnya menyesuaikan preferensi masing-masing, jika sudah terbiasa dengan snake-case, bisa gunakan bracket notation seperti diatas, namun alasan mengapa camelCase pada dasarnya akan lebih mempermudah adalah kita cukup memanggilnya seperti berikut:
<div className={alias.namaKelas}>

kembali lagi, menyesuaikan preferensi masing-masing

-->

<!-- agar struktur folder kita lebih clean / mengadaptasi konsep clean code, maka dari implementasi yang kita lakukan saat ini, kita bisa sedikit melakukan refactoring untuk struktur folder nya -->
<!-- kita akan coba menambahkan folder views yang berisi pendefinisian kode html nya, sementara itu untuk folder pages kita gunakan hanya untuk memanggil komponen views nya -->
<!-- tujuan dari melakukan refactoring ini ialah agar folder pages bisa lebih berfokus pada routing, dan folder views di inisialiasasikan untuk menyimpan file-file berkaitan dengan view, dengan begini folder pages bisa menghasilkan code yang jauh lebih seamless dan clean code karena murni fungsinya hanya sebagai routing -->

<!-- di Next.js juga memungkinkan kita untuk melakukan styling secara inline / langsung pada elemen html nya, untuk aturan penulisannya adalah sebagai berikut:
- menggunakan double kurung kurawal
- di inisialisasi sebagai, key: value
- penggunaan key yang lebih dari satu kata harus pake camelCase
<h1 style={{ color: red, fontWeight: bold }}>
 -->


 <!-- pemanfaatan SASS pada Next.js -->
 <!-- SASS sendiri merupakan singkatan dari Syntactically Awesome StyleSheet yang memungkinkan kita untuk menggunakan variabel, aturan bersarang, inline-import, dan sebagainya. -->
 <!-- untuk penggunaan SASS sendiri agar bisa digunakan didalam project ini, kita perlu menginstalnya via npm terlebih dahulu:

 npm i --save-dev sass
  -->
<!-- penamaan file SASS juga tidak sembarangan, ketika kita mencoba mendefinisikan sebuah file sass, maka ekstensi file nya adalah .scss -->
<!-- untuk meng-get style dari file SASS/.scss yang telah didefinisikan, kita bisa langsung import dengan cara:

import '@/styles/namaFile.scss'

kemudian jika kita ingin menggunakannya pada module/file css lainnya, kita perlu mengganti/rename nama file css ini menjadi .scss
setelahnya, untuk memanggil sass nya kita bisa menggunakan fungsi map-get, misal

.container {
    background-color: map-get($map: $schema, $key: primary)

    atau, kita bisa menyingkatnya sebagai berikut:
    background-color: map-get($schema, primary)
}
 -->


 <!-- Kemudian jika kita ingin menginstal tailwind pada project ini, sebenarnya cara paling disarankannya ialah dengan membaca dokumentasi di halaman resmi nya, mengingat perkembangan versi tailwind ini cukup masif dan selalu update versi di setiap periode nya, sehingga tidak memungkinkan jika harus dijelaskan secara step-by-step nya dengan rinci dan tidak bisa jadi patokan bahwa seterusnya cara/step-by-step nya seperti ini. -->
 <!-- Nah namun, menurut versi tailwindcss v4.3 (versi paling baru saat ini), berikut panduannya -->

 <!-- 1. jalankan instruksi berikut pada terminal -->
 npm install tailwindcss @tailwindcss/postcss postcss

 <!-- 2. buat file dengan nama postcss.config.mjs di root folder project kita dan tambahkan plugin `@tailwindcss/postcss` kedalam file configurasi postcss tersebut -->
 const config = {
  plugins: {
    "@tailwindcss/postcss": {},
  },
};

export default config;

<!-- 3. tambahkan @import ke dalam ./app/globals.css -->
@import "tailwindcss";

<!-- 4. jalankan npm run dev pada terminal -->
npm run dev
